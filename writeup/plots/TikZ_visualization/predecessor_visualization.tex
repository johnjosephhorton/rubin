% Three-row version of the comparative-advantage example. The benchmark sits in
% the middle; the two chaining channels branch above and below it, so the red
% arrows show the same focal step k flipping from manual to automated either way.
%   top    -- successor cheap to verify: {k, k+1} becomes one AI chain
%   middle -- benchmark: k manual, k+1 augmented (comparative advantage holds)
%   bottom -- reliable but expensive predecessor: {k-1, k, k+1} becomes one chain
% Colors follow Figures 1-2 (manual=gray, automated=green, augmented=orange;
% an AI chain is grouped by a blue dashed box).
\centering
\begin{adjustbox}{max width=\linewidth}
\begin{tikzpicture}[>=latex,
  box/.style={rectangle, rounded corners, draw, minimum width=1.9cm, minimum height=0.8cm, align=center, font=\small},
  manual/.style={fill=gray!10},
  automated/.style={fill=green!30},
  augmented/.style={fill=orange!30},
  idx/.style={draw=none, font=\scriptsize},
  ann/.style={draw=none, font=\scriptsize\itshape, text=black!55, align=center},
  rowlab/.style={draw=none, font=\footnotesize\itshape, align=right},
  cost/.style={draw=none, font=\footnotesize, align=left},
  chainbox/.style={draw=blue, rectangle, rounded corners, dashed, inner sep=0.20cm, line width=1pt}]

  % ---------------- middle row: benchmark ----------------
  \node[box, manual] (m1) {Manual};
  \node[box, augmented, right=0.6cm of m1] (m2) {Augmented};
  \draw[->, thick] (m1) -- (m2);
  \node[cost, anchor=west] at ($(m2.east)+(0.9,0)$) {Total cost $\approx 10.4$};

  % ---------------- top row: successor channel ----------------
  \node[box, automated, above=1.10cm of m1] (t1) {Automated};
  \node[box, augmented, right=0.6cm of t1] (t2) {Augmented};
  \draw[->, thick] (t1) -- (t2);
  \node[chainbox, fit=(t1)(t2)] (chaint) {};
  \node[idx, above=0.42cm of t1] {Step $k$};
  \node[idx, above=0.42cm of t2] {Step $k{+}1$};
  \node[ann] at ($(t2.south)+(0,-0.45)$) {cheap to verify};
  \node[cost, anchor=west] at ($(t2.east)+(0.9,0)$) {Total cost $\approx 7.4$\\(vs.\ $8.2$ unchained)};

  % ---------------- bottom row: predecessor channel ----------------
  \node[box, automated, below=1.20cm of m1] (b1) {Automated};
  \node[box, automated, left=0.6cm of b1] (b0) {Automated};
  \node[box, augmented, right=0.6cm of b1] (b2) {Augmented};
  \draw[->, thick] (b0) -- (b1);
  \draw[->, thick] (b1) -- (b2);
  \node[chainbox, fit=(b0)(b1)(b2)] (chainb) {};
  \node[idx, above=0.42cm of b0] {Step $k{-}1$};
  \node[ann] at ($(b0.south)+(0,-0.62)$) {reliable but expensive};
  \node[cost, anchor=west] at ($(b2.east)+(0.9,0)$) {Total cost $\approx 15.6$\\(vs.\ $35.4$ unchained)};

  % ---------------- the focal step flips either way ----------------
  % Drawn between explicit endpoints rather than with shorten keys, so the same
  % coordinates can centre the labels on the visible shaft (see below).
  \coordinate (tA) at ($(m1.north)+(0,0.07)$);
  \coordinate (tB) at ($(t1.south)+(0,-0.18)$);
  \coordinate (bA) at ($(m1.south)+(0,-0.07)$);
  \coordinate (bB) at ($(b1.north)+(0,0.18)$);
  \draw[->, very thick, red!75!black] (tA) -- (tB);
  \draw[->, very thick, red!75!black] (bA) -- (bB);
  \node[font=\scriptsize, text=red!75!black, anchor=east]
    at ($(tA)!0.5!(tB)+(-0.1,0)$) {flips to AI-automated};
  \node[font=\scriptsize, text=red!75!black, anchor=west]
    at ($(bA)!0.5!(bB)+(0.1,0)$) {flips to AI-automated};

  % ---------------- row labels, left of the widest (bottom) row ----------------
  \node[rowlab, anchor=east] at ($(b0.west |- t1)+(-0.7,0)$) {Cheap Successor\\Automation Channel:};
  \node[rowlab, anchor=east] at ($(b0.west |- m1)+(-0.7,0)$) {Benchmark:};
  \node[rowlab, anchor=east] at ($(b0.west)+(-0.7,0)$) {Expensive Predecessor\\Automation Channel:};
\end{tikzpicture}
\end{adjustbox}
