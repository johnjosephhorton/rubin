% Predecessor example: with only steps 1-2, the focal step (step 1) is performed
% manually; introducing the AI-friendly predecessor (step 0) pulls step 1 into a
% single AI chain. Colors follow Figures 1-2 (manual=gray, automated=green,
% augmented=orange; an AI chain is grouped by a blue dashed box). Box dimensions
% and spacing match the fragmentation figures, so a step reads the same size here.
\centering
\begin{adjustbox}{max width=\linewidth}
\begin{tikzpicture}[>=latex,
  box/.style={rectangle, rounded corners, draw, minimum width=1.8cm, minimum height=0.8cm, align=center, font=\small},
  manual/.style={fill=gray!10},
  automated/.style={fill=green!30},
  augmented/.style={fill=orange!30},
  idx/.style={draw=none, font=\scriptsize},
  rowlab/.style={draw=none, font=\footnotesize\itshape, align=right},
  chainbox/.style={draw=blue, rectangle, rounded corners, dashed, inner sep=0.20cm, line width=1pt}]

  % row (a): steps 1 and 2 only -- step 1 manual, step 2 augmented (separate tasks)
  \node[box, manual] (a1) {Manual};
  \node[box, augmented, right=0.6cm of a1] (a2) {Augmented};
  \node[idx, above=1pt of a1] {Step 1};
  \node[idx, above=1pt of a2] {Step 2};
  \draw[->, thick] (a1) -- (a2);
  \node[font=\footnotesize, align=left, anchor=west] at ($(a2.east)+(0.8,0)$) {Total cost $\approx 10.44$\\ \scriptsize(best alternative: $14$)};

  % row (b): introduce step 0 -- all three chained, steps 0 and 1 automated
  \node[box, automated, below=1.7cm of a1] (b1) {Automated};
  \node[box, automated, left=0.6cm of b1] (b0) {Automated};
  \node[box, augmented, right=0.6cm of b1] (b2) {Augmented};
  \draw[->, thick] (b0) -- (b1);
  \draw[->, thick] (b1) -- (b2);
  % blue dashed box: the three steps form a single AI-chain task (cf. Figures 1-2)
  \node[chainbox, fit=(b0)(b1)(b2)] (chain) {};
  \node[idx] at ($(b0.north)+(0,0.36)$) {Step 0};
  \node[font=\footnotesize, text=blue!60!black] at ($(chain.south)+(0,-0.30)$) {single task (AI chain)};
  \node[font=\footnotesize, align=left, anchor=west] at ($(b2.east)+(0.8,0)$) {Total cost $\approx 23.39$\\ \scriptsize(best alternative: $25.44$)};

  % row labels, both anchored to the left of the wider (lower) row
  \node[rowlab, anchor=east] at ($(b0.west |- a1)+(-0.6,0)$) {Steps 1--2 alone:};
  \node[rowlab, anchor=east] at ($(b0.west)+(-0.6,0)$) {Adding Step~0:};

  % step 1 flips from manual to automated once the predecessor is introduced
  \draw[->, very thick, red!75!black] (a1.south) -- (b1.north);
  \node[font=\scriptsize, text=red!75!black, anchor=west] at ($(a1.south)!0.5!(b1.north)+(0.15,0)$) {flips to AI-automated};
\end{tikzpicture}
\end{adjustbox}
